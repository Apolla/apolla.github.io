## 元祖
```swift

let error = (404, "Not Found")
print("code is\(error.0) message is \(error.1)")
  

let (code,msg) = error
print("code is\(code) message is \(msg)")

  

let httpResponse = (code:200,msg:"请求成功")
print("code is\(httpResponse.code) message is \(httpResponse.msg)")
```
